\newcommand{\series}[3]{%
  \addplot+[run pair,#1,forget plot] table[x=n,y=#2,y error=#2err]{#3};}
% Paint the total, then its lower component; apply the observed range at the total.
\newcommand{\stackpairbar}[7]{%
  \addplot[ybar,bar width=2.7mm,bar shift=0pt,draw=#1,fill=#5,line width=.4pt,forget plot]
    table[x expr={\thisrow{n}+(#2)},y=#3]{#7};
  \addplot[ybar,bar width=2.7mm,bar shift=0pt,draw=#1,fill=#6,line width=.4pt,forget plot]
    table[x expr={\thisrow{n}+(#2)},y=#4]{#7};
  \addplot[no marks,draw=none,forget plot,error bars/.cd,y dir=both,y explicit,
    error mark=none,error bar style={draw=black,line width=.6pt}]
    table[x expr={\thisrow{n}+(#2)},y=#3,y error=#3err]{#7};}
